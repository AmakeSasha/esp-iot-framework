40  /**
41  * @defgroup eif_mdns mDNS Configuration
42  * @brief Network discovery and mDNS management
43  *
44  * @note This module is available when Kconfig option `CONFIG_EIF_ENABLE_MDNS`
45  * is enabled and you include this line at the beginning of the file.:
46  * @code{c}
47  * #include <esp_iot_framework_core_mdns.h>
48  * @endcode
49  *
50  * This module allows you to configure mDNS. mDNS enables the device to be
51  * discovered via a human-readable name (e.g., `my-device.local`) instead of an
52  * IP address.
53  *
54  * The framework automatically appends a unique identifier based on
55  * the MAC address to prevent network collisions (e.g., common hostname set:
56  * `esp32.local`, the framework turns this into `esp32-1a2b3c.local` on a
57  * device with MAC `XX:XX:XX:1A:2B:3C`).
58  *
59  * @note In case the mDNS hostname is not set, `device` will be used as the
60  * common hostname. To disable mDNS, turn off Kconfig option
61  * `CONFIG_EIF_ENABLE_MDNS`.
62  *
63  * If you frequently need to enable and disable mDNS during development,
64  * you can use the following construct:
65  * @code{c}
66  * #include <mdns.h>
67  * #include <esp_err.h>
68  * #include <esp_iot_framework_core.h>
69  * #ifdef CONFIG_EIF_ENABLE_MDNS
70  * #include <esp_iot_framework_core_mdns.h>
71  * #endif
72  *
73  * // Not necessary, but nice :)
74  * #ifdef CONFIG_EIF_ENABLE_MDNS
75  * static const mdns_txt_item_t my_txt_records[] = {
76  * {"friendly_name", "Kitchen Main Light"},
77  * {"room", "Kitchen"},
78  * {"hardware", "ESP32-S3-DevKit"},
79  * {"v", "1.0.4"}
80  * };
81  * #endif
82  *
83  * void app_main(void) {
84  * ESP_ERROR_CHECK(eif_core_initialize());
85  * ESP_ERROR_CHECK(eif_nvs_initialize());
86  *
87  * #ifdef CONFIG_EIF_ENABLE_MDNS
88  * ESP_ERROR_CHECK(eif_set_mdns("my-esp", "Smart Controller"));
89  * ESP_ERROR_CHECK(eif_set_mdns_records(my_txt_records, 4));
90  * #endif
91  *
92  * // Further code...
93  * }
94  * @endcode
95  * @{
96  */

134  * ESP_ERROR_CHECK(eif_core_initialize());

147  /**
148  * @brief Set custom TXT records for mDNS service discovery.
149  *
150  * TXT records allow you to broadcast additional metadata about the device
151  * (e.g., firmware version, model, or status). The framework creates an
152  * internal copy of the structure, so the original data's lifecycle no
153  * longer matters after the call.
154  *
155  * @note This function must be called after `eif_core_initialize()` but before
156  * `eif_wifi_initialize()`.
157  *
158  * @warning The handler array length must be equal to `txt_records_count`. If
159  * the count exceeds the actual array length, a `Buffer Overread` will
160  * occur. If the count is less than the actual length, error
161  * `ESP_ERR_INVALID_ARG` will return.
162  * <br>
163  * If the array is accessible by value at the function call site, you
164  * can use the following code to automatically calculate the length:
165  * @code{c}
166  * sizeof(array) / sizeof(array[0])
167  * @endcode
168  *
169  * @param txt_records Array of mDNS TXT items. The length should be
170  * <code>#EIF_MDNS_TXT_RECORDS_MAX_COUNT</code>.
171  * @param txt_records_count Number of elements in the array.
172  *
173  * @return
174  * - `ESP_OK`: Records applied.
175  * - `ESP_ERR_INVALID_ARG`: The pointer `key` or `value` from
176  * `txt_records` is `NULL`.
177  * - `ESP_ERR_INVALID_SIZE`: The `txt_records` length is less than
178  * `txt_records_count`.
179  *
180  * Example of use:
181  * @code{c}
182  * #include <mdns.h>
183  * #include <esp_err.h>
184  * #include <esp_iot_framework_core.h>
185  * #include <esp_iot_framework_core_mdns.h>
186  *
187  * static const mdns_txt_item_t my_txt_records[] = {
188  * {"friendly_name", "Kitchen Main Light"},
189  * {"room", "Kitchen"},
190  * {"hardware", "ESP32-S3-DevKit"},
191  * {"v", "1.0.4"}
192  * };
193  *
194  * void app_main(void) {
195  * ESP_ERROR_CHECK(eif_core_initialize());
196  * ESP_ERROR_CHECK(eif_nvs_initialize());
197  * ESP_ERROR_CHECK(eif_set_mdns_records(my_txt_records, 4));
198  *
199  * // Further code...
200  * }
201  * @endcode
202  */
203  esp_err_t eif_set_mdns_records(
204  const mdns_txt_item_t txt_records[EIF_MDNS_TXT_RECORDS_MAX_COUNT],
205  size_t txt_records_count
206  );
